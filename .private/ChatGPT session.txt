"""

It is fine for a toy, but it does not demonstrate the RAG shape clearly. RAG is more naturally:

```text id="dwvr6c"
retrieve only the relevant policies at runtime
```

That retrieval step is what your BeeAI tool should demonstrate.

## Tool output becomes context

A tool result is not “outside” the prompt in any meaningful LLM sense. Once the tool returns data, the agent framework usually feeds that result back into the model’s conversation state so the model can use it.

So conceptually:

```text id="r6bppn"
Tool invocation retrieves data.
Retrieved data becomes context.
The model generates using that context.
```

This is why “tool or context?” is not an either/or. The tool is the retrieval mechanism. The returned policies are the retrieved context.

## 2. `company_travel_policy_tool.py`

```python
import json
import re
from pathlib import Path
from typing import Any

from pydantic import BaseModel, Field

from beeai_framework.context import RunContext
from beeai_framework.emitter import Emitter
from beeai_framework.tools import StringToolOutput, Tool, ToolRunOptions


class CompanyTravelPolicyToolInput(BaseModel):
    travel_phrase: str = Field(
        description=(
            "A short natural-language travel phrase, such as "
            "'take a plane to Rome', 'catch the ferry to Paris', "
            "or 'go by train and car to Sydney'."
        )
    )


class CompanyTravelPolicyTool(
    Tool[CompanyTravelPolicyToolInput, ToolRunOptions, StringToolOutput]
):
    name = "CompanyTravelPolicyLookup"

    description = (
        "Retrieves company travel policy statements relevant to a travel phrase. "
        "Use this when the user asks about travel and company travel policies may apply."
    )

    input_schema = CompanyTravelPolicyToolInput

    def __init__(
        self,
        policy_file: str | Path = "travel_policies.json",
        options: dict[str, Any] | None = None,
    ) -> None:
        super().__init__(options)
        self.policy_file = Path(policy_file)

    def _create_emitter(self) -> Emitter:
        return Emitter.root().child(
            namespace=["tool", "company", "travel_policy"],
            creator=self,
        )

    async def _run(
        self,
        input: CompanyTravelPolicyToolInput,
        options: ToolRunOptions | None,
        context: RunContext,
    ) -> StringToolOutput:
        data = self._load_policy_data()

        tag_aliases: dict[str, list[str]] = data.get("tag_aliases", {})
        policies: list[dict[str, Any]] = data.get("policies", [])

        matched_tags = self._extract_tags(input.travel_phrase, tag_aliases)
        matched_policies = self._find_matching_policies(matched_tags, policies)

        result = self._format_result(
            travel_phrase=input.travel_phrase,
            matched_tags=matched_tags,
            matched_policies=matched_policies,
        )

        return StringToolOutput(result=result)

    def _load_policy_data(self) -> dict[str, Any]:
        if not self.policy_file.exists():
            return {
                "tag_aliases": {},
                "policies": [],
            }

        with self.policy_file.open("r", encoding="utf-8") as file:
            return json.load(file)

    def _extract_tags(
        self,
        travel_phrase: str,
        tag_aliases: dict[str, list[str]],
    ) -> list[str]:
        words = set(re.findall(r"\b[a-zA-Z]+\b", travel_phrase.lower()))

        matched_tags: list[str] = []

        for tag, aliases in tag_aliases.items():
            normalized_aliases = {alias.lower() for alias in aliases}

            if words.intersection(normalized_aliases):
                matched_tags.append(tag)

        return matched_tags

    def _find_matching_policies(
        self,
        matched_tags: list[str],
        policies: list[dict[str, Any]],
    ) -> list[dict[str, Any]]:
        matched_tag_set = set(matched_tags)

        if not matched_tag_set:
            return []

        return [
            policy
            for policy in policies
            if matched_tag_set.intersection(set(policy.get("tags", [])))
        ]

    def _format_result(
        self,
        travel_phrase: str,
        matched_tags: list[str],
        matched_policies: list[dict[str, Any]],
    ) -> str:
        if not matched_tags:
            return (
                f"No company travel policy tags were found for: {travel_phrase!r}. "
                "No specific company travel policies apply."
            )

        if not matched_policies:
            return (
                f"Detected travel policy tags: {', '.join(matched_tags)}.\n"
                "No matching company travel policy statements were found."
            )

        lines = [
            f"Detected travel policy tags: {', '.join(matched_tags)}.",
            "Relevant company travel policies:",
        ]

        for policy in matched_policies:
            policy_id = policy.get("id", "unknown-policy")
            statement = policy.get("statement", "")
            lines.append(f"- {policy_id}: {statement}")

        return "\n".join(lines)
```
